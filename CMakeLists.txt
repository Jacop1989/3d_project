cmake_minimum_required(VERSION 3.16)
project(3d_project C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_compile_options(-Wall -Wextra -Wshadow -Wconversion -Wpedantic -fno-exceptions)

if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  add_compile_options(-O0 -g)
  include(CheckCCompilerFlag)
  check_c_compiler_flag("-fsanitize=address,undefined" HAS_SAN)
  if(HAS_SAN)
    add_compile_options(-fsanitize=address,undefined)
    add_link_options(-fsanitize=address,undefined)
  endif()
else()
  add_compile_options(-O2 -DNDEBUG)
endif()

file(GLOB_RECURSE SRC CONFIGURE_DEPENDS "src/*.c" "3d_*.c")
add_executable(3d_project ${SRC})

set(SRC_NO_MAIN ${SRC})
list(REMOVE_ITEM SRC_NO_MAIN ${CMAKE_SOURCE_DIR}/src/app/main.c)

target_include_directories(3d_project PRIVATE include src/app src/math src/render src/platform src/sketch src/core)

if (WIN32)
  target_link_libraries(3d_project PRIVATE gdi32 user32 kernel32)
else()
  target_link_libraries(3d_project PRIVATE X11 m)
endif()

add_executable(test_runner tests/test_runner.c ${SRC_NO_MAIN})
target_include_directories(test_runner PRIVATE include src/app src/math src/render src/platform src/sketch src/core)
if (WIN32)
  target_link_libraries(test_runner PRIVATE gdi32 user32 kernel32)
else()
  target_link_libraries(test_runner PRIVATE X11 m)
endif()
